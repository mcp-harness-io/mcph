1"""Find out what authentication an HTTP MCP server asks for. 

2 

3A server that requires OAuth answers a request without a (valid) token with 401 

4and points clients at its protected resource metadata (RFC 9728): in the 

5``resource_metadata`` parameter of the ``WWW-Authenticate`` challenge, or at a 

6well-known URL. The challenge can also say why a token was refused (``error``, 

7``error_description``) and which scope is needed (``scope``, RFC 6750). This 

8module reads that information. The only credential it ever sends is the bearer 

9token it is given, and only to the MCP endpoint itself. 

10""" 

11 

12from __future__ import annotations 

13 

14import asyncio 

15import re 

16from dataclasses import dataclass, replace 

17from urllib.parse import urlsplit 

18 

19import httpx2

21from mcph import __version__ 

22 

23_INITIALIZE_REQUEST = { 

24 "jsonrpc": "2.0", 

25 "id": 1, 

26 "method": "initialize", 

27 "params": { 

28 "protocolVersion": "2025-11-25", 

29 "capabilities": {}, 

30 "clientInfo": {"name": "mcph", "version": __version__}, 

31 }, 

32} 

33 

34# key="quoted value" or key=token, as used in WWW-Authenticate challenges. 

35_CHALLENGE_PARAM = re.compile( 

36 r'([A-Za-z][\w-]*)=(?:"((?:[^"\\]|\\.)*)"|([^\s,"]+))' 

37) 

38 

39_WELL_KNOWN = "/.well-known/oauth-protected-resource" 

40 

41 

42@dataclass(frozen=True) 

43class AuthInfo: 

44 """What an HTTP server says about the authentication it requires.""" 

45 

46 challenge: str | None 

47 """The ``WWW-Authenticate`` header of the 401 (or 403) response, if any.""" 

48 metadata_url: str | None = None 

49 """Where the protected resource metadata was found, if it was.""" 

50 authorization_servers: tuple[str, ...] = () 

51 scopes: tuple[str, ...] = () 

52 """The scopes the server's metadata lists as supported.""" 

53 status: int = 401 

54 error: str | None = None 

55 """The challenge's ``error``, e.g. ``invalid_token`` or ``insufficient_scope``.""" 

56 error_description: str | None = None 

57 required_scope: str | None = None 

58 """The challenge's ``scope``: what the server asks a token to have."""

61async def probe_auth( 

62 url: str, 

63 *, 

64 timeout: float, 

65 token: str | None = None, 

66 transport: httpx2.AsyncBaseTransport | None = None, 

67) -> AuthInfo | None: 

68 """Return what the server says about its auth if it refuses us, else ``None``. 

69 

70 Sends one MCP ``initialize`` request to ``url``, with ``Authorization: Bearer 

71 <token>`` if a token is given. A 401 counts as a refusal; so does a 403 whose 

72 challenge says ``insufficient_scope`` when a token was sent. Every other 

73 outcome (another status, a network error, a timeout) gives ``None``: this only 

74 explains a failure and must never hide it. ``transport`` is a test seam. 

75 """ 

76 try: 

77 async with asyncio.timeout(timeout): 

78 async with httpx2.AsyncClient( 

79 follow_redirects=True, timeout=timeout, transport=transport 

80 ) as http: 

81 return await _probe(http, url, token) 

82 except (httpx2.HTTPError, TimeoutError, ValueError): 

83 return None

86async def _probe( 

87 http: httpx2.AsyncClient, url: str, token: str | None 

88) -> AuthInfo | None: 

89 headers = {"Accept": "application/json, text/event-stream"} 

90 if token: 

91 headers["Authorization"] = f"Bearer {token}" 

92 response = await http.post(url, json=_INITIALIZE_REQUEST, headers=headers) 

93 

94 challenge = response.headers.get("www-authenticate") 

95 params = _challenge_params(challenge) 

96 if not _is_refusal(response.status_code, params, has_token=bool(token)): 

97 return None 

98 

99 info = AuthInfo( 

100 challenge=challenge, 

101 status=response.status_code, 

102 error=params.get("error"), 

103 error_description=params.get("error_description"), 

104 required_scope=params.get("scope"), 

105 ) 

106 # The metadata is public: it is fetched without the token, which must not 

107 # be sent anywhere but to the MCP endpoint. 

108 for candidate in _metadata_candidates(url, challenge): 

109 metadata = await _fetch_json_object(http, candidate) 

110 if metadata is not None: 

111 return replace( 

112 info, 

113 metadata_url=candidate, 

114 authorization_servers=_strings(metadata.get("authorization_servers")), 

115 scopes=_strings(metadata.get("scopes_supported")), 

116 ) 

117 return info

120def _is_refusal(status: int, params: dict[str, str], *, has_token: bool) -> bool: 

121 if status == 401: 

122 return True 

123 return status == 403 and has_token and params.get("error") == "insufficient_scope" 

124

126def _metadata_candidates(url: str, challenge: str | None) -> list[str]: 

127 """Where to look for the metadata: the challenge's pointer, then well-known URLs.""" 

128 candidates: list[str] = [] 

129 advertised = _challenge_params(challenge).get("resource_metadata") 

130 if advertised and urlsplit(advertised).scheme in ("http", "https"): 

131 candidates.append(advertised) 

132 

133 parts = urlsplit(url) 

134 root = f"{parts.scheme}://{parts.netloc}{_WELL_KNOWN}" 

135 if parts.path.rstrip("/"): 

136 candidates.append(root + parts.path.rstrip("/")) 

137 candidates.append(root) 

138 return list(dict.fromkeys(candidates))

141def _challenge_params(challenge: str | None) -> dict[str, str]: 

142 """The parameters of a ``WWW-Authenticate`` header; the first of each name wins.""" 

143 params: dict[str, str] = {} 

144 for match in _CHALLENGE_PARAM.finditer(challenge or ""): 

145 if match.group(2) is not None: # quoted string: undo backslash escapes 

146 value = re.sub(r"\\(.)", r"\1", match.group(2)) 

147 else: 

148 value = match.group(3) 

149 params.setdefault(match.group(1).lower(), value) 

150 return params

153async def _fetch_json_object(http: httpx2.AsyncClient, url: str) -> dict | None: 

154 try: 

155 response = await http.get(url, headers={"Accept": "application/json"}) 

156 data = response.json() if response.status_code == 200 else None 

157 except (httpx2.HTTPError, ValueError): 

158 return None 

159 return data if isinstance(data, dict) else None

162def _strings(value: object) -> tuple[str, ...]: 

163 if not isinstance(value, list): 

164 return () 

165 return tuple(item for item in value if isinstance(item, str))
